function buildEyewear(cfg) { while (glassesContainer.children.length) { glassesContainer.remove(glassesContainer.children[0]); } const root = new THREE.Group(); const frameMat = new THREE.MeshStandardMaterial({ color: cfg.frameColor, metalness: cfg.metalness, roughness: cfg.roughness }); const lensMat = new THREE.MeshPhysicalMaterial({ color: cfg.lensColor, metalness: 0, roughness: 0.05, transmission: 0.85, transparent: true, opacity: 0.72, depthWrite: false }); // Calibrated human anatomical proportions const eyeSep = 0.285; // Tighter eye separation const rimRadius = 0.185; // Realistic lens size (centered over pupils) const rimThickness = 0.018; // Sleek modern frame rim // Central Nose Bridge const bridgeGeo = new THREE.CylinderGeometry(rimThickness * 0.7, rimThickness * 0.7, eyeSep * 0.75, 16); bridgeGeo.rotateZ(Math.PI / 2); const bridge = new THREE.Mesh(bridgeGeo, frameMat); bridge.position.set(0, 0.01, 0); root.add(bridge); if (cfg.style === "aviator") { const topBar = new THREE.Mesh(new THREE.BoxGeometry(eyeSep * 2 + rimRadius, 0.012, 0.014), frameMat); topBar.position.set(0, 0.09, 0); root.add(topBar); } [-eyeSep, eyeSep].forEach((x) => { let rimGeo, lensGeo; if (cfg.style === "aviator") { rimGeo = new THREE.TorusGeometry(rimRadius, 0.014, 12, 36); lensGeo = new THREE.CircleGeometry(rimRadius * 0.95, 32); } else if (cfg.style === "square") { rimGeo = new THREE.TorusGeometry(rimRadius, rimThickness, 14, 36); lensGeo = new THREE.CircleGeometry(rimRadius * 0.95, 32); } else { rimGeo = new THREE.TorusGeometry(rimRadius, rimThickness, 16, 40); lensGeo = new THREE.CircleGeometry(rimRadius * 0.95, 36); } const rim = new THREE.Mesh(rimGeo, frameMat); if (cfg.style === "aviator") rim.scale.set(1.05, 1.22, 1); if (cfg.style === "square") rim.scale.set(1.15, 0.92, 1); rim.position.set(x, 0, 0); root.add(rim); const lens = new THREE.Mesh(lensGeo, lensMat); if (cfg.style === "aviator") lens.scale.set(1.05, 1.22, 1); if (cfg.style === "square") lens.scale.set(1.15, 0.92, 1); lens.position.set(x, 0, 0.002); root.add(lens); }); // Temples (Arms) mounted at upper-outer corner with natural downward ear slope const armLen = 1.42; const armGeo = new THREE.BoxGeometry(0.014, 0.022, armLen); const armOffsetX = eyeSep + rimRadius; const armOffsetY = 0.035; // Lifted to upper rim hinge const leftArm = new THREE.Mesh(armGeo, frameMat); leftArm.position.set(-armOffsetX, armOffsetY, -armLen / 2); leftArm.rotation.x = -0.06; // Slopes down to top of ear leftArm.rotation.y = 0.03; root.add(leftArm); const rightArm = new THREE.Mesh(armGeo, frameMat); rightArm.position.set(armOffsetX, armOffsetY, -armLen / 2); rightArm.rotation.x = -0.06; rightArm.rotation.y = -0.03; root.add(rightArm); // Position: Lifted slightly on Y to center directly across your pupils root.position.set(0, 0.025, 0.22); glassesContainer.add(root); }
